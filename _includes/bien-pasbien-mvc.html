C# using System; using System.Collections.Generic; // Model class TaskModel { public List<string> Tasks = new List<string>(); public void AddTask(string task) { Tasks.Add(task); } public List<string> GetAll() { return Tasks; } } // View class TaskView { public static void Display(List<string> tasks) { for (int i = 0; i < tasks.Count; i++) { Console.WriteLine($"{i+1}. {tasks[i]}"); } } } // Controller class TaskController { private TaskModel model; private TaskView view; public TaskController(TaskModel m, TaskView v) { model = m; view = v; } public void AddNewTask(string task) { model.AddTask(task); TaskView.Display(model.GetAll()); } }

JavaScript function divideNumbers(dividend, divisor) { try { if (divisor === 0) { throw new Error("Division by zero is not allowed"); } return dividend / divisor; } catch (error) { console.error(error.message); throw error; } finally { console.log("Cleanup complete"); } }
Python def divide_numbers(dividend, divisor): try: if divisor == 0: raise ValueError("Division by zero is not allowed") return dividend / divisor except ValueError as error: print(error) raise finally: print("Cleanup complete")
C# static double DivideNumbers(double dividend, double divisor) { try { if (divisor == 0) { throw new ArgumentException("Division by zero is not allowed"); } return dividend / divisor; } catch (ArgumentException error) { Console.Error.WriteLine(error.Message); throw; } finally { Console.WriteLine("Cleanup complete"); } }
